# 블로그(django_blog_Enhanced)와 같은 HTML이 나오도록 버전을 맞춘다.
# 올리면 같은 마크다운이 다른 HTML로 나오고, 블로그 쪽은 이 조합에서
# 벗어나면 빌드가 깨진다(블로그 requirements.txt 주석 참고).
markdown==3.4.1
pymdown-extensions==9.11

# 코드 하이라이팅을 서버에서 인라인 스타일로 박기 위해 필요.
# Blogger에는 CSS 클래스를 붙일 곳이 없다.
Pygments>=2.14

# 로컬 웹 UI 전용. converter/ 패키지는 Flask를 import하지 않는다 —
# 나중에 블로그(Django)로 이식할 때 폴더째 옮기기 위한 제약이다.
Flask>=3.0

# 역방향(HTML → 마크다운) 전용 — converter/to_markdown.py.
# 깨진 HTML·비정상 중첩을 버텨야 해서 표준 파서로는 부족하다.
# converter/__init__.py에서 import하지 않으므로, 정방향만 쓰는 곳
# (블로그가 converter/를 복사해 간다)에는 없어도 된다.
beautifulsoup4>=4.12

# 서버 저장(도구별 문서) 전용 — Neon Postgres.
# Vercel 함수는 디스크가 없어 SQLite를 쓸 수 없다(요청마다 다른 인스턴스일 수 있고
# 파일시스템도 사라진다). converter/ 패키지는 이 의존성을 import하지 않는다 —
# 블로그가 그 폴더를 복사해 가는데 저장 기능은 이 사이트에만 있다.
psycopg[binary]>=3.1
